public class GeometricProgression extends ArithmeticProgression
{
	public GeometricProgression(long start, long step)
	{
		super(start, step);
	}
	public GeometricProgression() {}//implicity calls parent constructor
	
	@Override //no override would catch it as a compiler error if you didn't use it
	protected void advance()
	{
		current *= step;
	}
}
